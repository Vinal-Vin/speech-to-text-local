# Windows dictation client.
# Deliberately few dependencies, all shipping prebuilt wheels -- no compiler
# is ever needed, which matters on a locked-down machine where build tools
# may not be installable.

pynput==1.7.7          # global push-to-talk hotkey (no admin needed on Windows)
sounddevice==0.5.1     # microphone capture via PortAudio
numpy==2.2.1           # audio buffers
httpx==0.28.1          # HTTP to the ASR container
pywin32==308 ; sys_platform == "win32"   # clipboard (CF_UNICODETEXT)
pystray==0.19.5        # system tray icon
Pillow==11.1.0         # tray icon rendering
tomli==2.2.1 ; python_version < "3.11"   # tomllib is stdlib from 3.11
